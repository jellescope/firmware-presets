#$ OPTION BEGIN (UNCHECKED): Lantian
    # -- ADDER: For the Lantian FURYFOSD IBUS --
    serial 2 2048 115200 57600 0 115200
    serial 5 64 115200 57600 0 115200
    
    set serialrx_provider = IBUS
    set rssi_channel = 14
    set dshot_bidir = ON
    set motor_pwm_protocol = DSHOT300
    set motor_poles = 14
    set gyro_1_align_yaw = 1800
    set craft_name = Lantian

    #set vtx power levels
    vtxtable powerlevels 4
    vtxtable powervalues 1 2 3 4
    vtxtable powerlabels 25 200 600 800

    # apply power setting for three levels with switch on controller (AUX 3)
    vtx 0 2 0 0 1 900 1200
    vtx 1 2 0 0 2 1300 1700
    vtx 2 2 0 0 4 1800 2100
#$ OPTION END
